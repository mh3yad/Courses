02 / 09
if

الشرط البسيط

الـ if بتخليك تنفذ كود معين بس لو شرط معين اتحقق. لو الشرط مش true، الكود اللي جواها بيتجاهل خالص.

if_simple.cpp
int score = 65;

if (score >= 50) {
    cout << "ناجح";
}
// score = 65، الشرط true، هيطبع "ناجح"
03 / 09
if with else

بديل لو الشرط مش true

لو عايز تنفذ حاجة تانية في حالة إن الشرط مكانش صح، بتضيف else بعد الـ if.

if_else.cpp
int score = 40;

if (score >= 50) {
    cout << "ناجح";
} else {
    cout << "راسب";
}
// score = 40، الشرط false، هيطبع "راسب"
04 / 09
if · else if · else

أكتر من شرط مرتبين

لو عايز تفرق بين أكتر من حالة (زي حروف التقييم)، بتحط else if بين الـ if والـ else. الـ ++C بيفحص الشروط بالترتيب وياخد أول واحد true.

grade_letter.cpp
int score = 85;

if (score >= 90) {
    cout << "A";
} else if (score >= 80) {
    cout << "B";
} else if (score >= 70) {
    cout << "C";
} else {
    cout << "F";
}
// score = 85، هيطبع "B"
05 / 09
Nested if

if جوه if

تقدر تحط if جوه if تاني، عشان تتحقق من شرط ثانوي بس لو الشرط الأساسي اتحقق الأول.

nested_if.cpp
int score = 95;

if (score >= 50) {
    cout << "ناجح";

    if (score >= 90) {
        cout << " ومتفوق";
    }
}
// الشرط الداخلي بيتفحص بس لو الطالب أصلا ناجح
06 / 09
اتنين if منفصلين

مختلفين عن else if

هنا كل if بيتفحص لوحده من غير ما يعتمد على التاني — ممكن الاتنين يتنفذوا مع بعض في نفس الوقت، عكس الـ else if اللي بتوقف عند أول شرط true.

two_if.cpp
int score = 92;
bool hasAttendance = true;

if (score >= 50) {
    cout << "ناجح";
}

if (score >= 90 && hasAttendance) {
    cout << " - مؤهل للمنحة";
}
// الشرط التاني بيتفحص لوحده، مش مربوط بالأول
07 / 09
مثال شامل

كل الحالات مع بعض

دلوقتي خلينا نجمع كل أنواع الـ if في برنامج واحد لطالب فيه score.

full_example.cpp
#include <iostream>
using namespace std;

int main() {
    int score = 92;
    bool hasAttendance = true;

    // if · else if · else للحرف
    if (score >= 90) {
        cout << "Grade: A" << endl;
    } else if (score >= 80) {
        cout << "Grade: B" << endl;
    } else {
        cout << "Grade: F" << endl;
    }

    // nested if للتفوق
    if (score >= 50) {
        cout << "ناجح" << endl;
        if (score >= 90) {
            cout << "ومتفوق" << endl;
        }
    }

    // if منفصل للمنحة
    if (score >= 90 && hasAttendance) {
        cout << "مؤهل للمنحة" << endl;
    }

    return 0;
}
08 / 09
Revision

اللي لازم يفضل في دماغك

if بتنفذ كود بس لو الشرط true.
else بتنفذ بديل لو الشرط false.
else if بتفرق بين أكتر من حالة، وبتوقف عند أول شرط true.
Nested if يعني if جوه if تاني لشرط ثانوي.
اتنين if منفصلين بيتفحصوا مستقلين عن بعض، ممكن يتنفذوا مع بعض.
اختار النوع المناسب على حسب عدد الحالات والعلاقة بينهم.
